(library (relay)
  (export my-let1 my-let2 my-let3)
  (import (chezscheme))

  (define-syntax my-let1
    (syntax-rules ()
      [(_ ((var val)) body)
       (let ((var val)) body)]))

  (define-syntax my-let2
    (syntax-rules ()
      [(_ ((var val)) body)
       (my-let1 ((var val)) body)]))

  (define-syntax my-let3
    (syntax-rules ()
      [(_ ((var val)) body)
       (my-let2 ((var val)) body)])))
